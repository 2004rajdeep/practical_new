    use Illuminate\Database\Schema\Blueprint;
    use Illuminate\Support\Facades\Schema;

    return new class extends Migration {
        public function up(): void
        {
            Schema::create('posts', function (Blueprint $table) {
                $table->id();
                $table->foreignId('user_id')->constrained()->onDelete('cascade');
                $table->string('title');
                $table->text('content');

        public function down(): void
        {
            Schema::dropIfExists('posts');
        }
    };

    <?php

    namespace App\Http\Controllers;

    use App\Models\Post;
    use Illuminate\Http\Request;

    class PostController extends Controller
    {
        public function index()
        {
            $posts = Post::latest()->get();
            return view('posts.index', compact('posts'));
        }

        public function create()
        {
            return view('posts.create');
        }

        public function store(Request $request)
        {
            $validated = $request->validate([
                'title' => 'required|max:255',
                'content' => 'required|string',
                'status' => 'required|in:draft,published',
            ]);

            Post::create($validated);
            return redirect()->route('posts.index')->with('success', 'Post created successfully!');
        }

        public function show(Post $post)
        {
            return view('posts.show', compact('post'));
        }

        public function edit(Post $post)
        {
            return view('posts.edit', compact('post'));
        }

        public function update(Request $request, Post $post)
        {
            $validated = $request->validate([
                'title' => 'required|max:255',
                'content' => 'required|string',
                'status' => 'required|in:draft,published',
            ]);

            $post->update($validated);
            return redirect()->route('posts.index')->with('success', 'Post updated successfully!');
        }

        public function destroy(Post $post)
        {
            $post->delete();
            return redirect()->route('posts.index')->with('success', 'Post deleted successfully!');
        }
    }

    The actual migration used in this session:
    Schema::create('tasks', function (Blueprint $table) {
        $table->id();
        $table->foreignId('user_id')->constrained()->onDelete('cascade');
        $table->string('title');
        $table->text('description')->nullable();
        $table->enum('priority', ['low', 'medium', 'high'])->default('medium');
        $table->boolean('completed')->default(false);

    MODEL: app/Models/Task.php
    <?php

    namespace App\Models;

    use Illuminate\Database\Eloquent\Model;
    use Illuminate\Database\Eloquent\Relations\BelongsTo;

    class Task extends Model
    {
        protected $fillable = [
            'title',
            'description',
            'priority',
            'completed',
        ];

        protected function casts(): array
        {
            return ['completed' => 'boolean'];
        }

        public function user(): BelongsTo
        {
            return $this->belongsTo(User::class);
        }
    }

    USER MODEL: add this method inside the existing app/Models/User.php class:
    public function tasks()
    {
        return $this->hasMany(Task::class);
    }

    Do not replace the entire User.php file. Add the method inside the existing User class, before its final closing brace.

    CONTROLLER: app/Http/Controllers/TaskController.php
    <?php

    namespace App\Http\Controllers;

    use App\Models\Task;
    use Illuminate\Http\Request;

    class TaskController extends Controller
    {
        public function index()
        {
            $tasks = auth()->user()->tasks()->latest()->get();
            return view('tasks.index', compact('tasks'));
        }

        public function create()
        {
            return view('tasks.create');
        }

        public function store(Request $request)
        {
            $validated = $request->validate([
                'title' => 'required|max:255',
                'description' => 'nullable|string',
                'priority' => 'required|in:low,medium,high',
            ]);

            auth()->user()->tasks()->create($validated);
            return redirect()->route('tasks.index')->with('success', 'Task created successfully!');
        }

        public function show(Task $task)
        {
            $this->authorizeTask($task);
            return view('tasks.show', compact('task'));
        }

        public function edit(Task $task)
        {
            $this->authorizeTask($task);
            return view('tasks.edit', compact('task'));
        }

        public function update(Request $request, Task $task)
        {
            $this->authorizeTask($task);

            $validated = $request->validate([
                'title' => 'required|max:255',
                'description' => 'nullable|string',
                'priority' => 'required|in:low,medium,high',
                'completed' => 'sometimes|boolean',
            ]);

            $validated['completed'] = $request->boolean('completed');
            $task->update($validated);

            return redirect()->route('tasks.index')->with('success', 'Task updated successfully!');
        }

        public function destroy(Task $task)
        {
            $this->authorizeTask($task);
            $task->delete();
            return redirect()->route('tasks.index')->with('success', 'Task deleted successfully!');
        }

        private function authorizeTask(Task $task): void
        {
            abort_unless($task->user_id === auth()->id(), 403);
        }
    }

A command is typed into PowerShell/Terminal from the project root, where the artisan file exists.

paste the complete code, and save. Do not paste PHP code into PowerShell.

3. ADD A SINGLE LINE OR METHOD
When a section says "ADD THIS LINE" or "ADD THIS METHOD", do not erase the whole file.
For a method, paste it INSIDE the existing class, before the class's final closing brace }.
For an import, add it near the other use statements at the top.

4. DATABASE MIGRATIONS
Create a migration only once. Before making another, run:
php artisan migrate:status
Never run php artisan migrate:fresh on a database that contains work you need; it drops tables.

5. ROUTES
Add each route only once in routes/web.php. Keep use statements near the top, after <?php.
Do not leave any characters or blank lines before <?php in PHP files.

6. BLADE FILES
Create each named .blade.php file in the exact folder given. Blade code goes in the file, not PowerShell.

PRACTICAL 16: FINAL LARAVEL PROJECT
============================================================

Aim:
Build a complete Laravel project using authentication, database tables, relationships, CRUD and a clean interface.

The lab manual allows one of these project ideas:
- Student Result Management System
- Library Book Management System
- Online Quiz/MCQ Application
- Event Registration Portal
- Simple E-Commerce Product Catalog

Suggested fast option:
Reuse the existing Student Task Manager as a demo foundation, or create a new project only if your instructor requires a separate final project.

Fresh project commands (only if a separate project is required):
cd C:\xampp\htdocs
composer create-project laravel/laravel my-project
cd my-project
composer require laravel/breeze --dev
php artisan breeze:install blade
npm.cmd install
npm.cmd run build

Database:
1. Create a database in phpMyAdmin.
2. Update .env with DB_CONNECTION=mysql, DB_HOST=127.0.0.1, DB_PORT=3306, DB_DATABASE=your_database_name, DB_USERNAME=root, DB_PASSWORD=.
3. Run:
php artisan config:clear
php artisan migrate

Generate resources (example: library books):
php artisan make:model Book -m
php artisan make:controller BookController --resource
php artisan make:migration create_books_table
NOTE: Do not run both make:model Book -m and make:migration create_books_table for the same table. The first command already creates the migration. Use one approach only.

Example migration body (inside the generated migration's up() method):
Schema::create('books', function (Blueprint $table) {
    $table->id();
    $table->foreignId('user_id')->constrained()->cascadeOnDelete();
    $table->string('title');
    $table->string('author');
    $table->string('isbn')->nullable();
    $table->timestamps();
});

After saving migration:
php artisan migrate

Example Book model: app/Models/Book.php
REPLACE THE FILE with:
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Book extends Model
{
    protected $fillable = ['title', 'author', 'isbn', 'user_id'];
}

Example relationship (ADD THIS METHOD inside the existing User class):
public function books()
{
    return $this->hasMany(Book::class);
}

Example resource route (ADD ONCE in routes/web.php):
use App\Http\Controllers\BookController;
Route::resource('books', BookController::class)->middleware('auth');

Generate CRUD controller method skeleton:
php artisan make:controller BookController --resource
Implement index/create/store/show/edit/update/destroy methods as demonstrated in Practical 7 and Practical 8. Create Blade files under resources/views/books/.
Then verify:
php artisan route:list --path=books
php artisan migrate:status
php artisan optimize:clear
php artisan serve --port=8011

Open:
http://127.0.0.1:8011/books

Final checks:
- Registration/login works.
- Database tables exist.
- Create, read, update, delete all work.
- Routes are registered.
- Views are readable.
- Validate form inputs.
- Do not call the project "deployed" unless it is actually hosted on a remote server. Running php artisan serve is local development.
