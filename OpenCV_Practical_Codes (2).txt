============================================================
PRACTICAL 6: CORNER DETECTION USING HARRIS CORNER DETECTOR
============================================================
import cv2
import numpy as np

img = cv2.imread('image.jpg')
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
dst = cv2.cornerHarris(np.float32(gray), 2, 3, 0.04)
img[dst > 0.01 * dst.max()] = [0, 0, 255]

cv2.imshow('Harris Corners', img)
cv2.waitKey(0)
cv2.destroyAllWindows()

PRACTICAL 7: FEATURE DETECTION AND MATCHING USING ORB
============================================================
import cv2

a = cv2.imread('image.jpg', 0)
b = cv2.imread('image2.jpg', 0)

orb = cv2.ORB_create()
k1, d1 = orb.detectAndCompute(a, None)
k2, d2 = orb.detectAndCompute(b, None)

matches = cv2.BFMatcher(cv2.NORM_HAMMING, True).match(d1, d2)
out = cv2.drawMatches(a, k1, b, k2, matches, None)

cv2.imshow('ORB Feature Matching', out)
cv2.waitKey(0)
cv2.destroyAllWindows()


============================================================
PRACTICAL 8: IMAGE SEGMENTATION USING THRESHOLDING AND K-MEANS
============================================================
import cv2
import numpy as np

img = cv2.imread('image.jpg')
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

_, binary = cv2.threshold(gray, 127, 255, cv2.THRESH_BINARY)
adaptive = cv2.adaptiveThreshold(
    gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
    cv2.THRESH_BINARY, 11, 2
)

Z = np.float32(img.reshape((-1, 3)))
criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 10, 1.0)
_, labels, centers = cv2.kmeans(
    Z, 3, None, criteria, 3, cv2.KMEANS_RANDOM_CENTERS
)
seg = np.uint8(centers[labels.flatten()].reshape(img.shape))

cv2.imshow('Binary Threshold', binary)
cv2.imshow('Adaptive Threshold', adaptive)
cv2.imshow('K-Means Segmentation', seg)
cv2.waitKey(0)
cv2.destroyAllWindows()


============================================================
PRACTICAL 9: FACE DETECTION USING HAAR CASCADE CLASSIFIER
============================================================
import cv2

img = cv2.imread('image.jpg')
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
face = cv2.CascadeClassifier(
    cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
)

for x, y, w, h in face.detectMultiScale(gray, 1.1, 5):
    cv2.rectangle(img, (x, y), (x + w, y + h), (0, 255, 0), 2)

cv2.imshow('Face Detection', img)
cv2.waitKey(0)
cv2.destroyAllWindows()


============================================================
PRACTICAL 10: MOTION DETECTION USING BACKGROUND SUBTRACTION
============================================================
import cv2

cap = cv2.VideoCapture(0)
bg = cv2.createBackgroundSubtractorMOG2()

while True:
    ret, frame = cap.read()
    if not ret:
        break

    mask = bg.apply(frame)
    cv2.imshow('Motion Detection', mask)

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
