PRACTICAL 1: IMAGE GRAYSCALE
------------------------------------------------
import cv2
import matplotlib.pyplot as plt

img = cv2.imread(r"D:\sample_color_scene.png")
if img is None:
    print("Image not found!")
else:
    rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    plt.figure(figsize=(9, 4))
    plt.subplot(1, 2, 1); plt.imshow(rgb); plt.title("Original"); plt.axis("off")
    plt.subplot(1, 2, 2); plt.imshow(gray, cmap="gray"); plt.title("Grayscale"); plt.axis("off")
    plt.tight_layout(); plt.show()

PRACTICAL 3: ROTATE IMAGE BY 90, 180, AND 270 DEGREES
------------------------------------------------
import cv2
import matplotlib.pyplot as plt

img = cv2.imread(r"D:\sample_color_scene.png")
if img is None:
    print("Image not found!")
else:
    images = [
        ("Original", img),
        ("90 degrees", cv2.rotate(img, cv2.ROTATE_90_CLOCKWISE)),
        ("180 degrees", cv2.rotate(img, cv2.ROTATE_180)),
        ("270 degrees", cv2.rotate(img, cv2.ROTATE_90_COUNTERCLOCKWISE))
    ]
    plt.figure(figsize=(10, 8))
    for i, (title, im) in enumerate(images, 1):
        plt.subplot(2, 2, i)
        plt.imshow(cv2.cvtColor(im, cv2.COLOR_BGR2RGB))
        plt.title(title); plt.axis("off")
    plt.tight_layout(); plt.show()

PRACTICAL 10: RED COLOR DETECTION USING HSV
------------------------------------------------
import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread(r"D:\sample_color_scene.png")
if img is None:
    print("Image not found!")
else:
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    mask1 = cv2.inRange(hsv, np.array([0, 100, 80]), np.array([10, 255, 255]))
    mask2 = cv2.inRange(hsv, np.array([170, 100, 80]), np.array([180, 255, 255]))
    mask = cv2.bitwise_or(mask1, mask2)
    result = cv2.bitwise_and(img, img, mask=mask)
    plt.figure(figsize=(12, 4))
    plt.subplot(1, 3, 1); plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB)); plt.title("Original"); plt.axis("off")
    plt.subplot(1, 3, 2); plt.imshow(mask, cmap="gray"); plt.title("Red Mask"); plt.axis("off")
    plt.subplot(1, 3, 3); plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB)); plt.title("Detected Red"); plt.axis("off")
    plt.tight_layout(); plt.show()

PRACTICAL 13: ADD AND REMOVE SALT-AND-PEPPER NOISE
------------------------------------------------
import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread(r"D:\sample_color_scene.png")
if img is None:
    print("Image not found!")
else:
    noisy = img.copy()
    n = int(img.shape[0] * img.shape[1] * 0.03)
    y = np.random.randint(0, img.shape[0], n)
    x = np.random.randint(0, img.shape[1], n)
    noisy[y, x] = 255
    y = np.random.randint(0, img.shape[0], n)
    x = np.random.randint(0, img.shape[1], n)
    noisy[y, x] = 0
    clean = cv2.medianBlur(noisy, 5)
    plt.figure(figsize=(12, 4))
    plt.subplot(1, 3, 1); plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB)); plt.title("Original"); plt.axis("off")
    plt.subplot(1, 3, 2); plt.imshow(cv2.cvtColor(noisy, cv2.COLOR_BGR2RGB)); plt.title("Noisy"); plt.axis("off")
    plt.subplot(1, 3, 3); plt.imshow(cv2.cvtColor(clean, cv2.COLOR_BGR2RGB)); plt.title("Median Filter"); plt.axis("off")
    plt.tight_layout(); plt.show()

PRACTICAL 15: OBJECT DETECTION USING CONTOURS
------------------------------------------------
import cv2
import matplotlib.pyplot as plt

img = cv2.imread(r"D:\sample_color_scene.png")
if img is None:
    print("Image not found!")
else:
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    _, binary = cv2.threshold(gray, 127, 255, cv2.THRESH_BINARY)
    contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    output = img.copy()
    cv2.drawContours(output, contours, -1, (0, 255, 0), 2)
    plt.figure(figsize=(9, 4))
    plt.subplot(1, 2, 1); plt.imshow(binary, cmap="gray"); plt.title("Binary"); plt.axis("off")
    plt.subplot(1, 2, 2); plt.imshow(cv2.cvtColor(output, cv2.COLOR_BGR2RGB)); plt.title("Contours"); plt.axis("off")
    plt.tight_layout(); plt.show()
    print("Contours found:", len(contours))

PRACTICAL 16: EROSION AND DILATION
------------------------------------------------
import cv2
import matplotlib.pyplot as plt

img = cv2.imread(r"D:\sample_color_scene.png")
if img is None:
    print("Image not found!")
else:
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    _, binary = cv2.threshold(gray, 127, 255, cv2.THRESH_BINARY)
    kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (3, 3))
    erosion = cv2.erode(binary, kernel, iterations=1)
    dilation = cv2.dilate(binary, kernel, iterations=1)
    plt.figure(figsize=(12, 4))
    plt.subplot(1, 3, 1); plt.imshow(binary, cmap="gray"); plt.title("Binary"); plt.axis("off")
    plt.subplot(1, 3, 2); plt.imshow(erosion, cmap="gray"); plt.title("Erosion"); plt.axis("off")
    plt.subplot(1, 3, 3); plt.imshow(dilation, cmap="gray"); plt.title("Dilation"); plt.axis("off")
    plt.tight_layout(); plt.show()

PRACTICAL 18: FACE DETECTION WITH HAAR CASCADE
Place a face photo at D:\faces.jpg before running.
------------------------------------------------
import cv2
import matplotlib.pyplot as plt

img = cv2.imread(r"D:\faces.jpg")
if img is None:
    print("Put a face photo at D:\\faces.jpg and rerun.")
else:
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    cascade = cv2.CascadeClassifier(cv2.data.haarcascades + "haarcascade_frontalface_default.xml")
    faces = cascade.detectMultiScale(gray, scaleFactor=1.1, minNeighbors=5, minSize=(30, 30))
    for (x, y, w, h) in faces:
        cv2.rectangle(img, (x, y), (x+w, y+h), (0, 255, 0), 2)
    plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB))
    plt.title("Faces detected: " + str(len(faces)))
    plt.axis("off"); plt.show()

PRACTICAL 19: IMAGE DIFFERENCE
Place two images at D:\image1.png and D:\image2.png.
------------------------------------------------
import cv2
import matplotlib.pyplot as plt

img1 = cv2.imread(r"D:\image1.png")
img2 = cv2.imread(r"D:\image2.png")
if img1 is None or img2 is None:
    print("Place image1.png and image2.png in D:\\")
else:
    img2 = cv2.resize(img2, (img1.shape[1], img1.shape[0]))
    diff = cv2.absdiff(img1, img2)
    gray = cv2.cvtColor(diff, cv2.COLOR_BGR2GRAY)
    _, mask = cv2.threshold(gray, 30, 255, cv2.THRESH_BINARY)
    plt.figure(figsize=(12, 4))
    plt.subplot(1, 3, 1); plt.imshow(cv2.cvtColor(img1, cv2.COLOR_BGR2RGB)); plt.title("Image 1"); plt.axis("off")
    plt.subplot(1, 3, 2); plt.imshow(cv2.cvtColor(img2, cv2.COLOR_BGR2RGB)); plt.title("Image 2"); plt.axis("off")
    plt.subplot(1, 3, 3); plt.imshow(mask, cmap="gray"); plt.title("Difference"); plt.axis("off")
    plt.tight_layout(); plt.show()

PRACTICAL 20: GREEN LEAF SEGMENTATION
Best with a photo of a green leaf.
------------------------------------------------
import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread(r"D:\sample_color_scene.png")
if img is None:
    print("Image not found!")
else:
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    lower = np.array([35, 40, 40])
    upper = np.array([85, 255, 255])
    mask = cv2.inRange(hsv, lower, upper)
    segmented = cv2.bitwise_and(img, img, mask=mask)
    plt.figure(figsize=(12, 4))
    plt.subplot(1, 3, 1); plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB)); plt.title("Original"); plt.axis("off")
    plt.subplot(1, 3, 2); plt.imshow(mask, cmap="gray"); plt.title("Green Mask"); plt.axis("off")
    plt.subplot(1, 3, 3); plt.imshow(cv2.cvtColor(segmented, cv2.COLOR_BGR2RGB)); plt.title("Segmented"); plt.axis("off")
    plt.tight_layout(); plt.show()
