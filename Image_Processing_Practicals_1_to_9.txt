PRACTICAL 1: INTRODUCTION TO IMAGE PROCESSING TOOLBOX
=====================================================

AIM:
To introduce image-processing tools in Python using Pillow, NumPy and Matplotlib, and perform basic image operations.

CODE:
-----
from PIL import Image, ImageDraw
import numpy as np
import matplotlib.pyplot as plt

# Create a sample image (300 x 200 pixels)
img = Image.new("RGB", (300, 200), "white")
draw = ImageDraw.Draw(img)
draw.rectangle((30, 30, 130, 120), fill="red")
draw.ellipse((160, 40, 260, 140), fill="blue")

plt.figure(figsize=(6, 4))
plt.imshow(img)
plt.title("Sample Image")
plt.axis("off")
plt.show()

print("Format:", img.format)  # None because this image has not been saved to a file
print("Size:", img.size)
print("Mode:", img.mode)

gray = img.convert("L")
plt.figure(figsize=(6, 4))
plt.imshow(gray, cmap="gray")
plt.title("Grayscale Image")
plt.axis("off")
plt.show()

pixels = np.array(img)
print("Array shape:", pixels.shape)
print("First pixel (RGB):", pixels[0, 0])

EXPECTED OBSERVATION:
The image is displayed, dimensions and mode are printed, a grayscale version is displayed, and the RGB pixel array is shown.


PRACTICAL 2: OVERLAY AN IMAGE ON ANOTHER IMAGE IN PYTHON
=======================================================

AIM:
To place one image over another using Pillow.

CODE:
-----
from PIL import Image, ImageDraw
import matplotlib.pyplot as plt

background = Image.new("RGBA", (400, 250), (230, 240, 255, 255))
bg_draw = ImageDraw.Draw(background)
bg_draw.rectangle((0, 190, 400, 250), fill=(120, 180, 120, 255))

overlay = Image.new("RGBA", (140, 100), (0, 0, 0, 0))
ov_draw = ImageDraw.Draw(overlay)
ov_draw.ellipse((10, 10, 90, 90), fill=(255, 80, 50, 220))
ov_draw.rectangle((75, 25, 130, 75), fill=(255, 200, 0, 220))

result = background.copy()
result.alpha_composite(overlay, (220, 55))

plt.figure(figsize=(8, 4))
plt.imshow(result)
plt.title("Overlay Image on Background")
plt.axis("off")
plt.show()

result.save("practical2_overlay.png")

PRACTICAL 4: READ AN IMAGE, PLOT ITS HISTOGRAM, THEN DO HISTOGRAM EQUALIZATION
===============================================================================

AIM:
To read an image, plot its grayscale histogram, apply histogram equalization, and compare the results.

CODE:
-----
from PIL import Image, ImageDraw, ImageOps
import numpy as np
import matplotlib.pyplot as plt

# Generate a low-contrast grayscale sample image
sample = Image.new("L", (300, 200), 100)
draw = ImageDraw.Draw(sample)
draw.rectangle((30, 30, 130, 120), fill=110)
draw.ellipse((160, 40, 260, 140), fill=120)
sample.save("practical4_original.png")

img = Image.open("practical4_original.png").convert("L")
before_hist = img.histogram()

equalized = ImageOps.equalize(img)
after_hist = equalized.histogram()
equalized.save("practical4_equalized.png")

plt.figure(figsize=(10, 7))
plt.subplot(2, 2, 1)
plt.imshow(img, cmap="gray", vmin=0, vmax=255)
plt.title("Original Grayscale Image")
plt.axis("off")

plt.subplot(2, 2, 2)
plt.imshow(equalized, cmap="gray", vmin=0, vmax=255)
plt.title("After Histogram Equalization")
plt.axis("off")

plt.subplot(2, 2, 3)
plt.plot(range(256), before_hist)
plt.title("Histogram Before")
plt.xlabel("Pixel intensity")
plt.ylabel("Frequency")

plt.subplot(2, 2, 4)
plt.plot(range(256), after_hist)
plt.title("Histogram After")
plt.xlabel("Pixel intensity")
plt.ylabel("Frequency")

plt.tight_layout()
plt.show()
print("Saved practical4_original.png and practical4_equalized.png")

COMMENT:
Histogram equalization remaps intensity values to spread useful pixel levels over a wider range. Contrast may improve, especially for a low-contrast image. The resulting histogram is not necessarily perfectly flat, and results depend on the image.


PRACTICAL 5: CONVERT AN IMAGE INTO JPG FORMAT USING PILLOW
=========================================================

AIM:
To convert an image to JPEG format using Pillow.

CODE:
-----
from PIL import Image, ImageDraw
import matplotlib.pyplot as plt

img = Image.new("RGBA", (300, 200), (255, 255, 255, 255))
draw = ImageDraw.Draw(img)
draw.rectangle((30, 30, 140, 150), fill=(30, 150, 230, 255))
draw.ellipse((160, 40, 260, 140), fill=(240, 80, 80, 255))

# JPEG does not support transparency, so convert to RGB first
rgb_img = Image.new("RGB", img.size, "white")
rgb_img.paste(img, mask=img.getchannel("A"))
rgb_img.save("practical5_converted.jpg", format="JPEG", quality=95)

plt.imshow(Image.open("practical5_converted.jpg"))
plt.title("Converted JPG Image")
plt.axis("off")
plt.show()

print("Saved as practical5_converted.jpg")
print("Format:", Image.open("practical5_converted.jpg").format)


PRACTICAL 6: IMPLEMENT VARIOUS SMOOTHING SPATIAL FILTERS
=======================================================

AIM:
To apply mean, Gaussian, and median smoothing filters to an image and compare the results.

CODE:
-----
from PIL import Image, ImageFilter
import numpy as np
import matplotlib.pyplot as plt

rng = np.random.default_rng(7)
base = np.zeros((220, 300), dtype=np.uint8)
base[35:185, 40:260] = 180
noise = rng.integers(-35, 36, size=base.shape)
noisy = np.clip(base.astype(np.int16) + noise, 0, 255).astype(np.uint8)
img = Image.fromarray(noisy)

mean_img = img.filter(ImageFilter.BoxBlur(2))
gaussian_img = img.filter(ImageFilter.GaussianBlur(radius=2))
median_img = img.filter(ImageFilter.MedianFilter(size=3))

plt.figure(figsize=(10, 8))
plt.subplot(2, 2, 1)
plt.imshow(img, cmap="gray", vmin=0, vmax=255)
plt.title("Original Noisy Image")
plt.axis("off")

plt.subplot(2, 2, 2)
plt.imshow(mean_img, cmap="gray", vmin=0, vmax=255)
plt.title("Mean / Box Blur")
plt.axis("off")

plt.subplot(2, 2, 3)
plt.imshow(gaussian_img, cmap="gray", vmin=0, vmax=255)
plt.title("Gaussian Blur")
plt.axis("off")

plt.subplot(2, 2, 4)
plt.imshow(median_img, cmap="gray", vmin=0, vmax=255)
plt.title("Median Filter")
plt.axis("off")

plt.tight_layout()
plt.show()
print("Mean, Gaussian, and Median filters applied.")

PRACTICAL 8: EXTRACT IMAGE METADATA IN PYTHON
=============================================

AIM:
To extract basic image metadata such as format, dimensions, mode, and available EXIF information.

CODE:
-----
from PIL import Image, ImageDraw, ExifTags

img = Image.new("RGB", (320, 180), "lightblue")
draw = ImageDraw.Draw(img)
draw.ellipse((90, 30, 220, 160), fill="orange")
img.save("practical8_metadata_sample.png")

with Image.open("practical8_metadata_sample.png") as im:
    print("Format:", im.format)
    print("Dimensions (width, height):", im.size)
    print("Mode:", im.mode)
    print("File information:", im.info)

    exif = im.getexif()
    if exif:
        print("EXIF metadata:")
        for tag_id, value in exif.items():
            print(f"{ExifTags.TAGS.get(tag_id, tag_id)}: {value}")
    else:
        print("No EXIF metadata found in this generated sample image.")

NOTE:
A generated PNG usually has little or no EXIF metadata. To inspect camera details, replace the filename with a real photo that contains EXIF data. Not every image contains EXIF information.


PRACTICAL 9: IMAGE EDGE DETECTION USING PILLOW
==============================================

AIM:
To detect edges in an image using Pillow's FIND_EDGES filter.

CODE:
-----
from PIL import Image, ImageFilter, ImageDraw
import matplotlib.pyplot as plt

img = Image.new("RGB", (300, 200), "white")
draw = ImageDraw.Draw(img)
draw.rectangle((30, 30, 130, 170), fill="black")
draw.ellipse((160, 40, 270, 150), fill="gray")
draw.line((0, 100, 300, 100), fill="red", width=5)

edges = img.filter(ImageFilter.FIND_EDGES)

plt.figure(figsize=(9, 4))
plt.subplot(1, 2, 1)
plt.imshow(img)
plt.title("Original Image")
plt.axis("off")

plt.subplot(1, 2, 2)
plt.imshow(edges, cmap="gray")
plt.title("Detected Edges")
plt.axis("off")

plt.tight_layout()
plt.show()

edges.save("practical9_edges.png")
print("Saved as practical9_edges.png")
