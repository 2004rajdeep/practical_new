Aim:
Share a user name between components without passing props through every level.
No extra package is required.

Code for App.js:

import React, { createContext, useContext, useState } from 'react';
import {
  View, Text, TextInput, Pressable, StyleSheet
} from 'react-native';

const UserContext = createContext(null);

function Profile() {
  const { user, updateUser } = useContext(UserContext);
  const [newName, setNewName] = useState('');

  const handleUpdate = () => {
    if (newName.trim()) {
      updateUser(newName.trim());
      setNewName('');
    }
  };

  return (
    <View style={styles.card}>
      <Text style={styles.cardHeading}>Profile Component</Text>
      <Text style={styles.welcomeText}>Welcome, {user}</Text>
      <TextInput
        style={styles.input}
        placeholder="Enter a new name"
        value={newName}
        onChangeText={setNewName}
      />
      <Pressable style={styles.button} onPress={handleUpdate}>
        <Text style={styles.buttonText}>Update Global User</Text>
      </Pressable>
    </View>
  );
}

function Dashboard() {
  const { user } = useContext(UserContext);

  return (
    <View style={styles.card}>
      <Text style={styles.cardHeading}>Dashboard Component</Text>
      <Text style={styles.welcomeText}>
        Current logged-in user: {user}
      </Text>
    </View>
  );
}

export default function App() {
  const [user, setUser] = useState('John');
  const updateUser = (name) => setUser(name);

  return (
    <UserContext.Provider value={{ user, updateUser }}>
      <View style={styles.container}>
        <Text style={styles.heading}>Context API Example</Text>
        <Profile />
        <Dashboard />
      </View>
    </UserContext.Provider>
  );
}

PRACTICAL 12: API INTEGRATION USING FETCH
======================================================================

Aim:
Fetch posts from an API and display them using FlatList.
No extra package is required. An internet connection is required.

Code for App.js:

import React, { useEffect, useState } from 'react';
import {
  View, Text, FlatList, ActivityIndicator, Pressable, StyleSheet
} from 'react-native';

const API_URL = 'https://jsonplaceholder.typicode.com/posts';

export default function App() {
  const [posts, setPosts] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');

  const fetchPosts = async () => {
    try {
      setLoading(true);
      setError('');
      const response = await fetch(API_URL);
      if (!response.ok) {
        throw new Error(`Server returned status ${response.status}`);
      }
      const data = await response.json();
      setPosts(data.slice(0, 15));
    } catch (fetchError) {
      setError(fetchError.message || 'Unable to load the posts.');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchPosts();
  }, []);

  if (loading) {
    return (
      <View style={styles.centerContainer}>
        <ActivityIndicator size="large" />
        <Text style={styles.loadingText}>Loading posts...</Text>
      </View>
    );
  }

  if (error) {
    return (
      <View style={styles.centerContainer}>
        <Text style={styles.errorHeading}>Unable to Load Data</Text>
        <Text style={styles.errorText}>{error}</Text>
        <Pressable style={styles.reloadButton} onPress={fetchPosts}>
          <Text style={styles.reloadButtonText}>Try Again</Text>
        </Pressable>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      <Text style={styles.heading}>API Posts</Text>
      <FlatList
        data={posts}
        keyExtractor={(item) => item.id.toString()}
        showsVerticalScrollIndicator={false}
        renderItem={({ item }) => (
          <View style={styles.card}>
            <Text style={styles.postNumber}>Post #{item.id}</Text>
            <Text style={styles.title}>{item.title}</Text>
            <Text style={styles.body}>{item.body}</Text>
          </View>
        )}
      />
    </View>
  );
}

import React from 'react';

const validationSchema = Yup.object({
  name: Yup.string().trim().min(3, 'Name must contain at least 3 characters.').required('Name is required.'),
  email: Yup.string().trim().email('Enter a valid email address.').required('Email is required.'),
  password: Yup.string().min(6, 'Password must contain at least 6 characters.').required('Password is required.'),
});

export default function App() {
  const submitForm = (values, { resetForm }) => {
    const message = `Registration Successful\nName: ${values.name}\nEmail: ${values.email}`;
    if (typeof window !== 'undefined') window.alert(message);
    resetForm();
  };

      style={styles.container}
      behavior={Platform.OS === 'ios' ? 'padding' : undefined}
    >
      <Formik
        initialValues={{ name: '', email: '', password: '' }}

        onSubmit={submitForm}
      >
        {({ values, errors, touched, handleChange, handleBlur, handleSubmit, isValid }) => (
          <View style={styles.form}>
            <Text style={styles.heading}>Create Account</Text>

            <Text style={styles.label}>Name</Text>
            <TextInput
              style={[styles.input, touched.name && errors.name ? styles.invalidInput : null]}
              placeholder="Enter your name"
              value={values.name}
              onChangeText={handleChange('name')}
              onBlur={handleBlur('name')}
            />
            {touched.name && errors.name ? <Text style={styles.errorText}>{errors.name}</Text> : null}

            <Text style={styles.label}>Email</Text>
            <TextInput
              style={[styles.input, touched.email && errors.email ? styles.invalidInput : null]}
              placeholder="Enter your email"
              value={values.email}
              onChangeText={handleChange('email')}
              onBlur={handleBlur('email')}
              keyboardType="email-address"
              autoCapitalize="none"
            />
            {touched.email && errors.email ? <Text style={styles.errorText}>{errors.email}</Text> : null}

            <Text style={styles.label}>Password</Text>
            <TextInput

              placeholder="Enter your password"
              value={values.password}

              onBlur={handleBlur('password')}

            />
            {touched.password && errors.password ? <Text style={styles.errorText}>{errors.password}</Text> : null}

            <Pressable style={[styles.button, !isValid ? styles.disabledButton : null]} onPress={handleSubmit}>
              <Text style={styles.buttonText}>Register</Text>
            </Pressable>
          </View>
        )}
      </Formik>
    </KeyboardAvoidingView>
  );
}

IMPORTANT:
This practical requires a Firebase project and configuration. Create a project at
https://console.firebase.google.com, add a Web App, and enable Authentication >
Sign-in method > Email/Password. Copy the configuration values into firebaseConfig.js.
Never share private credentials. Firebase web configuration is not a service-account
private key, but keep account access and server secrets private.

Create a file named firebaseConfig.js in the project root:

import { getApp, getApps, initializeApp } from 'firebase/app';
import { getAuth } from 'firebase/auth';

const firebaseConfig = {
  apiKey: 'YOUR_API_KEY',
  authDomain: 'YOUR_PROJECT.firebaseapp.com',
  projectId: 'YOUR_PROJECT_ID',
  storageBucket: 'YOUR_PROJECT.appspot.com',
  messagingSenderId: 'YOUR_SENDER_ID',
  appId: 'YOUR_APP_ID',
};

const app = getApps().length === 0 ? initializeApp(firebaseConfig) : getApp();
export const auth = getAuth(app);

Replace the placeholder values with the configuration from your Firebase project.

Code for App.js:

import React, { useEffect, useState } from 'react';
import {
  View, Text, TextInput, Pressable, ActivityIndicator, StyleSheet
} from 'react-native';
import {
  createUserWithEmailAndPassword,
  onAuthStateChanged,
  signInWithEmailAndPassword,
  signOut,
} from 'firebase/auth';
import { auth } from './firebaseConfig';

export default function App() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [user, setUser] = useState(null);
  const [screenLoading, setScreenLoading] = useState(true);
  const [buttonLoading, setButtonLoading] = useState(false);
  const [isRegisterMode, setIsRegisterMode] = useState(false);
  const [message, setMessage] = useState('');

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (currentUser) => {
      setUser(currentUser);
      setScreenLoading(false);
    });
    return unsubscribe;
  }, []);

  const handleAuthentication = async () => {
    setMessage('');
    if (!email.trim()) {
      setMessage('Email is required.');
      return;
    }
    if (password.length < 6) {
      setMessage('Password must contain at least 6 characters.');
      return;
    }
    try {
      setButtonLoading(true);
      if (isRegisterMode) {
        await createUserWithEmailAndPassword(auth, email.trim(), password);
        setMessage('Account created successfully.');
      } else {
        await signInWithEmailAndPassword(auth, email.trim(), password);
      }
      setEmail('');
      setPassword('');
    } catch (error) {
      setMessage(error.message || 'Authentication failed.');
    } finally {
      setButtonLoading(false);
    }
  };

  const handleLogout = async () => {
    try {
      await signOut(auth);
    } catch (error) {
      setMessage(error.message || 'Logout failed.');
    }
  };

  if (screenLoading) {
    return (
      <View style={styles.centerContainer}>
        <ActivityIndicator size="large" />
        <Text>Checking authentication...</Text>
      </View>
    );
  }

  if (user) {
    return (
      <View style={styles.centerContainer}>
        <Text style={styles.heading}>Welcome</Text>
        <Text style={styles.userEmail}>{user.email}</Text>
        <Pressable style={styles.button} onPress={handleLogout}>
          <Text style={styles.buttonText}>Logout</Text>
        </Pressable>
        {message ? <Text>{message}</Text> : null}
      </View>
    );
  }

  return (
    <View style={styles.container}>
      <View style={styles.form}>
        <Text style={styles.heading}>{isRegisterMode ? 'Create Account' : 'Login'}</Text>
        <TextInput
          style={styles.input}

        </Pressable>

      </View>
    </View>
  );
}

import React, { useEffect, useState } from 'react';
import {
  View, Text, TextInput, Pressable, StyleSheet
} from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';

const PROFILE_KEY = 'studentProfile';

export default function App() {
  const [name, setName] = useState('');
  const [savedName, setSavedName] = useState('');
  const [message, setMessage] = useState('');

  const loadProfile = async () => {
    try {
      const storedValue = await AsyncStorage.getItem(PROFILE_KEY);
      if (storedValue) {
        const profile = JSON.parse(storedValue);
        setSavedName(profile.name);
      } else {
        setSavedName('');
      }
    } catch (error) {
      setMessage('Unable to load the profile.');
    }
  };

  const saveProfile = async () => {
    if (!name.trim()) {
      setMessage('Please enter a name.');
      return;
    }
    try {
      const profile = { name: name.trim(), savedAt: new Date().toISOString() };
      await AsyncStorage.setItem(PROFILE_KEY, JSON.stringify(profile));
      setSavedName(profile.name);
      setName('');
      setMessage('Profile saved locally.');
    } catch (error) {
      setMessage('Unable to save the profile.');
    }
  };

  const removeProfile = async () => {
    try {
      await AsyncStorage.removeItem(PROFILE_KEY);
      setSavedName('');
      setMessage('Saved profile removed.');
    } catch (error) {
      setMessage('Unable to remove the profile.');
    }
  };

  useEffect(() => {
    loadProfile();
  }, []);

  return (
    <View style={styles.container}>
      <Text style={styles.heading}>AsyncStorage Example</Text>
      <View style={styles.card}>
        <Text style={styles.label}>Enter Student Name</Text>
        <TextInput
          style={styles.input}
          placeholder="Enter name"
          value={name}
          onChangeText={setName}
        />
        <Pressable style={styles.saveButton} onPress={saveProfile}>
          <Text style={styles.buttonText}>Save Name</Text>
        </Pressable>
        <Pressable style={styles.loadButton} onPress={loadProfile}>
          <Text style={styles.buttonText}>Load Saved Name</Text>
        </Pressable>
        <Pressable style={styles.removeButton} onPress={removeProfile}>
          <Text style={styles.buttonText}>Remove Saved Name</Text>
        </Pressable>
      </View>
      {message ? <Text style={styles.message}>{message}</Text> : null}
      <View style={styles.resultCard}>
        <Text style={styles.resultLabel}>Saved Name</Text>
        <Text style={styles.resultValue}>{savedName || 'No name is currently saved.'}</Text>
      </View>
    </View>
  );
}
